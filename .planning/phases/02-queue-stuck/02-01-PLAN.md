**Step 1: Scan PID files and collect stuck processes**

```bash
local stuck_pids=()
local stuck_names=()
local stuck_runtimes=()
local stuck_staleness=()

for pid_file in "${LOG_DIR}"/gsd-*-pid; do
  [[ -f "${pid_file}" ]] || continue

  # Extract session name from path: /tmp/gsd-SESSIONNAME-pid → SESSIONNAME
  local session
  session="${pid_file##*/gsd-}"
  session="${session%-pid}"

  local status
  status=$(check_stuck "${session}")
  # check_stuck() sets globals STUCK_RUNTIME and STUCK_LOG_STALENESS

  if [[ "${status}" == "stuck" ]]; then
    local pid
    pid=$(< "${pid_file}")
    stuck_pids+=("${pid}")
    stuck_names+=("${session}")
    stuck_runtimes+=("${STUCK_RUNTIME}")
    stuck_staleness+=("${STUCK_LOG_STALENESS}")
  fi
  # not_running/not_found: check_stuck() already cleaned up stale PID files (§2.3.8, §5.9)
done
```

if [[ "${#stuck_pids[@]}" -eq 0 ]]; then

**Step 4: Display stuck processes table**

Print header: `print_header "Stuck Processes (threshold: ${STUCK_THRESHOLD_MIN}m)"`
Print table header: `print_table_header "%-8s %-25s %-12s %-15s" "PID" "SESSION" "RUNTIME" "LOG IDLE"`

Loop over stuck arrays:
```bash
local i
for (( i=0; i<${#stuck_pids[@]}; i++ )); do
  local short_session
  short_session=$(truncate_str "${stuck_names[${i}]}" 25)
  local runtime_fmt
  runtime_fmt=$(format_duration "${stuck_runtimes[${i}]}")
  local staleness_fmt
  staleness_fmt=$(format_duration "${stuck_staleness[${i}]}")
  printf '%s%-8s %-25s %-12s %-15s%s\n' \
    "${BOLD_RED}" \
    "${stuck_pids[${i}]}" \
    "${short_session}" \
    "${runtime_fmt}" \
    "${staleness_fmt}" \
    "${RESET}"
done
printf '\n'
```

**Step 5: Kill workflow (§2.3.5, §2.3.6, §2.3.7, §5.8)**

Only execute if `[[ "${kill_stuck}" == "true" ]]`.

Confirmation prompt (skip if `force=true`):
```bash
if [[ "${force}" == "false" ]]; then
  # Only prompt if stdin is a TTY (Pitfall #3 — don't hang in non-interactive)
  if [[ ! -t 0 ]]; then
    die "Cannot prompt for confirmation in non-interactive mode. Use --force to skip."
  fi
  printf '%sWill send SIGTERM to %d stuck process(es) listed above.%s\n' "${BOLD_RED}" "${#stuck_pids[@]}" "${RESET}"
  local confirm
  read -r -p "Kill these processes? [y/N]: " confirm
  if [[ "${confirm}" != "y" && "${confirm}" != "Y" ]]; then
    printf 'Aborted.\n'
    return 0
  fi
fi
```

Kill loop:
```bash
printf '\n'
for (( i=0; i<${#stuck_pids[@]}; i++ )); do
  local pid="${stuck_pids[${i}]}"
  local session="${stuck_names[${i}]}"
  # Re-verify alive before kill (§2.3.7) — process may have exited since we scanned
  if kill -0 "${pid}" 2>/dev/null; then
    kill "${pid}"  # SIGTERM only (§5.8) — never SIGKILL automatically
    printf '%sKilled PID %s (session: %s)%s\n' "${GREEN}" "${pid}" "${session}" "${RESET}"
    rm -f "${LOG_DIR}/gsd-${session}-pid"  # Clean up PID file after successful kill
  else
    printf '%sPID %s (session: %s) already exited%s\n' "${DIM}" "${pid}" "${session}" "${RESET}"
    rm -f "${LOG_DIR}/gsd-${session}-pid"  # Clean up stale PID file
  fi
done
```

**Implementation notes:**
- ALL local array declarations at top of function (before use): `local stuck_pids=()` etc.
- The `session` variable in the loop must be `local session` — but recall: `local` inside a function is scoped to function. The outer function scope is fine for this loop.
- Use `(( i=0; i < ...; i++ ))` style C-loop — cleaner for indexed array access than `for item in "${arr[@]}"` when parallel arrays are needed
- Remember `|| true` after `(( i++ ))` is NOT needed in a C-style `for` loop header — the increment in the third field is not checked by `set -e`
